## 树状数组

```c++
struct BIT {
    int n;
    vector<int> w, tmp; // tmp 为传入的待处理数组(数据拷贝)
    
    template<class V> BIT(V &in) : n(in.size() - 1), w(in.size()), chk(in) {}
    
    //单改区查
    void add(int x, int v) {
        for (; x <= n; x += x & -x) {
            w[x] += v;
        }
    }
    int rangeAsk(int l, int r) { 
        auto ask = [&](int x) {
            int ans = 0;
            for (; x > 0 ; x -= x & -x) {
                ans += w[x];
            }
            return ans;
        };
        return ask(r) - ask(l - 1);
    }
    
    //逆序对统计 - 核心
    int get() {
        vector<array<int, 2>> alls;
        for (int i = 1; i <= n; i++) {
            alls.push_back({chk[i], i});
        }
        sort(alls.rbegin(), alls.rend());
        int ans = 0;
        for (const auto &item : alls) {
            int idx = item[1];
            ans += ask(1, idx - 1);
            add(idx, 1);
        }
        return ans;
    }
};
```

- 时间复杂度：$O(n \log n)$
- 空间复杂度：$O(n)$（辅助临时数组）

```c++
void merge(int l, int r) {
    if(l >= r) return;
    
    int mid = (l + r) / 2;
    merge(l, mid);
    merge(mid + 1, r);

        else {
            tmp[k++] = a[j++];
            ans += mid - i + 1;
        }
    }
    
    while(i <= mid) tmp[k++] = a[i++];
    while(j <= r) tmp[k++] = a[j++];
    
    //update
    for(int p = l; p <= r; p++) a[p] = tmp[p];
}
```

- 时间复杂度：$O(n \log n)$
- 空间复杂度：$O(n)$（需要临时数组 `tmp`）
