void f(int v, int w, int cnt) {
    for (int c = 1; c <= cnt; c <<= 1) {
        val.push_back(c * v);      // 捆绑后的新价值
        weight.push_back(c * w);   // 捆绑后的新重量

    }

### 单调队列优化 ：

```c++
auto value = [&](int i, int j) -> int {
    return dp[j] - j / v[i] * w[i];  
};

for (int i = 0; i < n; i++) {
    int number = cnt[i];
    int cost = v[i];
    int add = w[i];
   
    for (int mod = 0; mod <= min(t, cost - 1); mod++) {
        deque<int> dq;
        for (int j = t - mod, c = 1; j >= 0 && c <= number; j -= cost, c++) {
            while (!dq.empty() && value(i, dq.back()) <= value(i, j)) {
                dq.pop_back();
            }
            dq.push_back(j);
        }
        
        for (int l = t - mod, r = l - cost * number; l >= 0; r -= cost, l -= cost) {
            if (r >= 0) {
                while (!dq.empty() && value(i, dq.back()) <= value(i, r)) {
                    dq.pop_back();
                }
                dq.push_back(r);
            } 

            dp[l] = value(i, dq.front()) + l / cost * add;

            if (dq.front() == l) dq.pop_front();
        }
    }
}
```

时间复杂度：**$O(n \times t)$** （其中 $t$ 为背包容量）
空间复杂度：**$O(t)$**




## 数学

### 位运算

#### 四种位运算及其运算律

设操作数为整数 $a,b,c$，符号说明：

- $\sim$：按位取反（一元运算）
- $ \wedge $：按位与
- $\mid$：按位或
- $\oplus$（代码写作 `^`）：按位异或

---

#### 一、按位取反 $\sim x$ 一元运算

不存在交换律、结合律，满足以下规律：

1. 双重否定律

$$\sim(\sim a) = a$$

2. 德摩根律（配合按位与、按位或）

$$\sim(a \wedge b) = (\sim a)\mid(\sim b),\quad \sim(a\mid b) = (\sim a) \wedge (\sim b)$$

3. 补码恒等式（计算机有符号整数特有）

$$\sim a = -a-1$$

---

#### 二、按位与 $ \wedge $（二元运算）

1. 交换律

$$a \wedge b = b \wedge a$$

2. 结合律

$$(a \wedge b) \wedge c = a \wedge (b \wedge c)$$

3. 幂等律

$$a \wedge a = a$$

4. 吸收律（搭配按位或）

$$a \wedge (a\mid b) = a$$

5. 分配律（对按位或分配）

$$a \wedge (b\mid c) = (a \wedge b)\mid(a \wedge c)$$

6. 零元 / 单位元性质

$$a \wedge 0 = 0,\quad a \wedge \text{全1掩码}=a$$

7. 互补律（搭配按位取反）

$$a \wedge (\sim a) = 0$$

---

#### 三、按位或 $\mid$（二元运算）

1. 交换律

$$a\mid b = b\mid a$$

2. 结合律

$$(a\mid b)\mid c = a\mid(b\mid c)$$

3. 幂等律

$$a\mid a = a$$

4. 吸收律（搭配按位与）

$$a\mid(a \wedge b) = a$$

5. 分配律（对按位与分配）

$$a\mid(b \wedge c) = (a\mid b) \wedge (a\mid c)$$

6. 零元 / 单位元性质

$$a\mid 0 = a,\quad a\mid\text{全1掩码}=\text{全1掩码}$$

7. 互补律（搭配按位取反）

$$a\mid(\sim a) = \text{全1掩码}$$

---

#### 四、按位异或 $\oplus$（代码符号 `^`，二元运算）

重要结论：异或不满足对 $ \wedge $、 $\mid$ 的分配律

1. 交换律

$$a \oplus b = b \oplus a$$

2. 结合律

$$(a \oplus b) \oplus c = a \oplus (b \oplus c)$$

3. 归零律（自身异或结果为 0）

$$a \oplus a = 0$$

4. 恒等律（异或 0 数值不变）

$$a \oplus 0 = a$$

5. 自逆可逆性质

若 $x = a \oplus b$，则 $a = x \oplus b,\ b = x \oplus a$

### 子集问题 - 二进制

## 一、 集合与集合的运算

| **集合论概念**     | **数学符号**        | **位运算写法**                    | **原理简述**                           |
| ------------- | --------------- | ---------------------------- | ---------------------------------- |
| **交集**        | $A \cap B$      | a & b                        | 两个集合都有的元素（两边都为 1 的位）               |
| **并集**        | $A \cup B$      | a \| b                       | 合并两个集合的所有元素（任一边为 1 的位）             |
| **对称差 / 异或**  | $A \triangle B$ | a ^ b                        | 存在于其中一个集合、但不同时存在的元素                |
| **差集 / 补集**   | $A \setminus B$ | a & ~b                       | 属于 $A$ 且不属于 $B$ 的元素（去掉 $B$ 中也有的元素） |
| **包含于（判断子集）** | $A \subseteq B$ | (a & b) == a 或 (a & ~b) == 0 | 检查 $A$ 的所有元素是否都在 $B$ 中             |

## 二、 元素与集合的单点操作

针对单个元素 $i$ 和集合 $s$ 的日常操作：

1. **查询元素**：判断 $i$ 是否在集合 $s$ 中
    
    - 写法：`(s >> i) & 1` 或 `(s & (1 << i)) != 0`
        
2. **添加元素**：将 $i$ 加入集合 $s$
    
    - 写法：`s | (1 << i)`
        
3. **删除元素**：将 $i$ 从集合 $s$ 中移除
    
    - 写法：`s & ~(1 << i)`
        

## 三、 常用底层技巧与内置函数

在写代码时，许多语言（如 C++, Python, Go）提供了高效的内置函数或经典按位技巧：

- **集合大小（统计 1 的个数）**：
    
    - 计算集合中有多少个元素，对应 C++ 中的 `__builtin_popcount(s)`（`long long` 用 `__builtin_popcountll`）。
        
- **Lowbit（最低位的 1 / 最小元素）**：
    
    - `s & -s`：能够直接提取出二进制中最低位的那个 `1` 及其后面的所有 `0`（常用于树状数组或遍历集合中的元素）。
        
- **获取集合的最大/最小元素**：
    
    - **最小元素**：可以通过计算尾零个数（如 C++ 的 `__builtin_ctz`）或直接利用 `Lowbit`。
        
    - **最大元素**：可以通过计算二进制最高位（如 C++ 的 `__lg(s)`）。
    
- **计算元素二进制的长度**： `bit_width(1ULL * x)（c++20）`或 `64 - __builtin_clzll(x)（c++14）` ;

## 马拉车

```c++
string longestPalindrome(string s) {
    if (s.empty()) return "";

    // 1. 预处理字符串：在字符间及首尾插入特殊字符（如 '#' 和边界符），统一奇偶回文处理
    string t = "^";
    for (char c : s) {
        t += "#";
        t += c;
    }
    t += "#$";

    int n = t.size();
    vector<int> p(n, 0); // p[i] 表示以 i 为中心的回文半径（包含 i 本身）
    int id = 0, mx = 0;  // id 为当前右边界最大的回文子串的中心，mx 为其右边界 (id + p[id])
    int max_len = 0;     // 记录原字符串中最长回文子串的长度
    int center_index = 0;// 记录最长回文子串在新字符串 t 中的中心位置

    // 2. 核心计算过程
    for (int i = 1; i < n - 1; i++) {
        // 利用对称性初始化 p[i]
        p[i] = (i < mx) ? min(mx - i, p[2 * id - i]) : 1;

        // 尝试中心扩展
        while (t[i - p[i]] == t[i + p[i]]) {
            p[i]++;
        }

        // 如果新回文串的右边界超过了 mx，更新 id 和 mx
        if (i + p[i] > mx) {
            id = i;
            mx = i + p[i];
        }

        // 记录全局最大回文长度及中心
        if (p[i] - 1 > max_len) {
            max_len = p[i] - 1;
            center_index = i;
        }
    }

    // 3. 根据 t 中的中心和长度还原回原字符串中的起始位置
    int start = (center_index - max_len) / 2;
    return s.substr(start, max_len);
}
```

// 输入 nums，返回每个元素右侧第一个比它大的元素，没有则返回 -1

vector<int> nextGreaterElement(vector<int>& nums) {
    int n = nums.size();
    vector<int> ans(n, -1);
    stack<int> st; // 存元素值（也可以存下标，看需求）

    // 从右往左遍历
    for (int i = n - 1; i >= 0; --i) {
        // 弹出栈顶比当前元素小的（它们不可能是任何左边元素的答案）
        while (!st.empty() && nums[st.top()] <= nums[i]) {
            st.pop();
        }
        // 栈顶就是第一个比 nums[i] 大的元素
        if (!st.empty()) {
            ans[i] = nums[st.top()];
        }
        // 当前元素入栈
        st.push(i);
    }
    return ans;
}

void dfs(int x, int p) {
    deep[x] = deep[p] + 1;
    st[0][x] = p;
    for (int i = 1; i < 20; i++) {
        st[i][x] = st[i - 1][st[i - 1][x]];
    }
    for (int e = head[x]; e; e = nxt[e]) {
        dfs(to[e], x);
    }
}

int Lca(int x, int y) {
    if (deep[x] > deep[y]) swap(x, y);

    // 将 x 和 y 调到同一深度
    for (int i = 19; i >= 0; i--) {
        if (deep[st[i][y]] >= deep[x]) {
            y = st[i][y];
        } 
    }
    if (x == y) return x;

    for (int i = 19; i >= 0; i--) {
        int dx = st[i][x];
        int dy = st[i][y];
        if (dx != dy) {
            x = dx;
            y = dy;
        }
    }
    return st[0][x];
}

void rebuildkruskalTree() {
    sort(edges.begin(), edges.end());
    for (auto e : edges) {
        int u = e.u;

        int w = e.w;

        weight[++node] = w;
        fa[u] = fa[v] = node;
        fa[node] = node; 

        addedge(node, u);
        addedge(node, v);
    }
}
